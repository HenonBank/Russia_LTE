#include <iostream>
#include <string>

using namespace std;

// Объявление структуры, которой не хватает на доске
struct Student {
    string name;
    int age;
    double gpa; // средний балл
};

int main() {
    // Фрагмент с доски: инициализация массива структур
    Student group[3] = {
        {"Иван", 20, 4.3},
        {"Мария", 19, 5.0},
        {"Петр", 18, 3.8}
    };

    // Фрагмент с доски: вывод имен студентов в цикле
    for (int i = 0; i < 3; i++) {
        cout << "Студент: " << group[i].name << endl;
    }

    return 0;
}
